# Android and host scene/JNI checks share the actual independent menu backend.
# It preserves room and actor resources while an original item is examined.
get_directory_property(task_scene_targets BUILDSYSTEM_TARGETS)
foreach(task_scene_target IN LISTS task_scene_targets)
    # Host contracts contain real assert checks, also in a Release build.
    if(NOT ANDROID AND task_scene_target MATCHES "_tests$")
        target_compile_options(${task_scene_target} PRIVATE -UNDEBUG)
    endif()
    get_target_property(task_scene_sources ${task_scene_target} SOURCES)
    if("scene_gl.cpp" IN_LIST task_scene_sources)
        target_sources(${task_scene_target} PRIVATE subscreen_item_gl.cpp subscreen_portrait_gl.cpp display_brightness_gl.cpp depth_of_field_gl.cpp item_effects_gl.cpp room_effect_cloth_gl.cpp room_filter_gl.cpp room_shadows_gl.cpp)
        if(NOT "radio_term_gl.cpp" IN_LIST task_scene_sources)
            target_sources(${task_scene_target} PRIVATE radio_term_gl.cpp)
        endif()
    endif()
    if("world_effects_gl.cpp" IN_LIST task_scene_sources AND NOT "source_glow_gl.cpp" IN_LIST task_scene_sources)
        target_sources(${task_scene_target} PRIVATE source_glow_gl.cpp)
    endif()
endforeach()
